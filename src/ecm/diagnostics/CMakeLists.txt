cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
    project(ecm_diagnostics LANGUAGES CXX)
endif()
get_filename_component(ECM_DIAGNOSTICS_REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(ECM_DIAGNOSTICS_GMP_ROOT "${ECM_DIAGNOSTICS_REPO}/third_party/gmp-zen3/dist" CACHE PATH "Diagnostic host GMP prefix")
set(ECM_DIAGNOSTICS_PARI_ROOT "${ECM_DIAGNOSTICS_REPO}/third_party/pari-2.17.3" CACHE PATH "Prepared PARI runtime and sources")
option(ECM_DIAGNOSTICS_LOCAL_PARI "Bundle the PARI DLL and SEA tables" ON)
if(NOT WIN32)
    message(STATUS "PARI DLL host is qualified on Win64 only; GP and explicit GMP remain available")
    return()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(ecm_diagnostics_host pari_host.cpp pari_native.cpp ../group_order/group_order.cpp)
set_target_properties(ecm_diagnostics_host PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
target_compile_features(ecm_diagnostics_host PRIVATE cxx_std_17)
target_compile_definitions(ecm_diagnostics_host PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_compile_options(ecm_diagnostics_host PRIVATE /utf-8 /EHsc)
target_include_directories(ecm_diagnostics_host PRIVATE "${ECM_DIAGNOSTICS_REPO}/src" "${ECM_DIAGNOSTICS_REPO}/include" "${ECM_DIAGNOSTICS_GMP_ROOT}/include")
target_link_libraries(ecm_diagnostics_host PRIVATE "${ECM_DIAGNOSTICS_GMP_ROOT}/lib/gmp.lib")
target_link_options(ecm_diagnostics_host PRIVATE /MANIFEST:EMBED "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/host.manifest")
set_property(TARGET ecm_diagnostics_host APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/host.manifest")
set(_pari_args)
if(NOT ECM_DIAGNOSTICS_LOCAL_PARI)
    list(APPEND _pari_args --without-pari)
endif()
add_custom_target(ecm_diagnostics_inputs
    COMMAND "${Python3_EXECUTABLE}" "${ECM_DIAGNOSTICS_REPO}/tools/packaging/diagnostics_runtime.py"
        --freeze-inputs "${CMAKE_BINARY_DIR}/diagnostics-inputs.json" --gmp "${ECM_DIAGNOSTICS_GMP_ROOT}" --pari "${ECM_DIAGNOSTICS_PARI_ROOT}" ${_pari_args}
    VERBATIM)
add_dependencies(ecm_diagnostics_host ecm_diagnostics_inputs)
add_custom_target(ecm_diagnostics_runtime ALL
    COMMAND "${Python3_EXECUTABLE}" "${ECM_DIAGNOSTICS_REPO}/tools/packaging/diagnostics_runtime.py"
        --binary "$<TARGET_FILE:ecm_diagnostics_host>" --gmp "${ECM_DIAGNOSTICS_GMP_ROOT}" --pari "${ECM_DIAGNOSTICS_PARI_ROOT}" ${_pari_args}
    DEPENDS ecm_diagnostics_host
    VERBATIM)
add_executable(ecm_diagnostics_probe EXCLUDE_FROM_ALL "${ECM_DIAGNOSTICS_REPO}/tools/test/diagnostics_probe.cpp" order_analysis.cpp ../group_order/group_order.cpp)
target_compile_features(ecm_diagnostics_probe PRIVATE cxx_std_17)
target_compile_definitions(ecm_diagnostics_probe PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(ecm_diagnostics_probe PRIVATE /utf-8 /EHsc)
target_include_directories(ecm_diagnostics_probe PRIVATE "${ECM_DIAGNOSTICS_REPO}/src" "${ECM_DIAGNOSTICS_REPO}/include" "${ECM_DIAGNOSTICS_GMP_ROOT}/include")
target_link_libraries(ecm_diagnostics_probe PRIVATE "${ECM_DIAGNOSTICS_GMP_ROOT}/lib/gmp.lib")
set_target_properties(ecm_diagnostics_probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
add_dependencies(ecm_diagnostics_probe ecm_diagnostics_runtime)
