cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
    project(ecm_group_order LANGUAGES CXX)
endif()
get_filename_component(ECM_GROUP_ORDER_REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(ECM_GROUP_ORDER_GMP_ROOT "${ECM_GROUP_ORDER_REPO}/third_party/gmp-zen3/dist" CACHE PATH "GMP installation prefix")
find_path(ECM_GROUP_ORDER_GMP_INCLUDE gmp.h PATHS "${ECM_GROUP_ORDER_GMP_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
find_library(ECM_GROUP_ORDER_GMP_LIBRARY NAMES gmp libgmp PATHS "${ECM_GROUP_ORDER_GMP_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
if(NOT ECM_GROUP_ORDER_GMP_INCLUDE STREQUAL "${ECM_GROUP_ORDER_GMP_ROOT}/include" OR
   NOT ECM_GROUP_ORDER_GMP_LIBRARY MATCHES "^${ECM_GROUP_ORDER_GMP_ROOT}/lib/")
    message(FATAL_ERROR "GMP cache differs from the requested prefix; use a fresh build directory")
endif()
add_library(ecm_group_order STATIC group_order.cpp)
target_compile_features(ecm_group_order PUBLIC cxx_std_17)
target_include_directories(ecm_group_order PUBLIC "${ECM_GROUP_ORDER_REPO}/src" "${ECM_GROUP_ORDER_GMP_INCLUDE}")
target_link_libraries(ecm_group_order PUBLIC "${ECM_GROUP_ORDER_GMP_LIBRARY}")
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    add_executable(group_order_probe "${ECM_GROUP_ORDER_REPO}/tools/test/group_order_probe.cpp")
    target_link_libraries(group_order_probe PRIVATE ecm_group_order)
    if(WIN32 AND EXISTS "${ECM_GROUP_ORDER_GMP_ROOT}/bin/gmp-10.dll")
        add_custom_command(TARGET group_order_probe POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "${ECM_GROUP_ORDER_GMP_ROOT}/bin/gmp-10.dll" "$<TARGET_FILE_DIR:group_order_probe>")
    endif()
    enable_testing()
    add_test(NAME cpu_group_order_self_test COMMAND group_order_probe --self-test)
endif()
